test_schema = '''"""
test_schema.py — Testes do JSON Schema do status do firmware.
"""

import jsonschema
import pytest


def test_schema_valido(schema_status):
    jsonschema.Draft7Validator.check_schema(schema_status)


def test_payload_valido(schema_status, payload_status_valido):
    jsonschema.validate(payload_status_valido, schema_status)


def test_payload_sem_campo_obrigatorio(schema_status, payload_status_valido):
    del payload_status_valido["temp_c"]
    with pytest.raises(jsonschema.ValidationError):
        jsonschema.validate(payload_status_valido, schema_status)


def test_payload_tipo_errado(schema_status, payload_status_valido):
    payload_status_valido["temp_c"] = "24.0"
    with pytest.raises(jsonschema.ValidationError):
        jsonschema.validate(payload_status_valido, schema_status)


def test_payload_estado_invalido(schema_status, payload_status_valido):
    payload_status_valido["estado"] = "ESTADO_QUE_NAO_EXISTE"
    with pytest.raises(jsonschema.ValidationError):
        jsonschema.validate(payload_status_valido, schema_status)


def test_payload_campo_extra(schema_status, payload_status_valido):
    payload_status_valido["campo_novo"] = 42
    with pytest.raises(jsonschema.ValidationError):
        jsonschema.validate(payload_status_valido, schema_status)


def test_payload_temp_fora_da_faixa(schema_status, payload_status_valido):
    payload_status_valido["temp_c"] = 150
    with pytest.raises(jsonschema.ValidationError):
        jsonschema.validate(payload_status_valido, schema_status)


def test_payload_umid_fora_da_faixa(schema_status, payload_status_valido):
    payload_status_valido["umid_pct"] = 150

        jsonschema.validate(payload_status_valido, schema_status)


def test_payload_ip_invalido(schema_status, payload_status_valido):

🛠️ Problema 2 — O teste test_upsert_item_nao_duplica está errado
O teste espera que, na segunda chamada, o item.update seja chamado. Mas o código está correto — ele chama item.update na segunda vez. O problema é que o teste lê o call_args_list[1] (a segunda chamada), mas a função upsert_dependent_item faz duas chamadas (item.get + item.update), então o índice [1] é o item.get, não o item.update.

A correção é no teste, não no código. Vamos ajustar o teste para verificar a última chamada:

O problema do CORS
O iframe é servido de localhost:3000, mas a API do Zabbix está em localhost:8080. Isso é origem cruzada (CORS), e o navegador pode bloquear o fetch().

Soluções:

Opção	Como funciona	Prós	Contras
A — Proxy reverso do Grafana	Adicionar um proxy no grafana.ini que redireciona /api/zabbix para http://zabbix-web:8080	CORS resolvido, sem mudar o Zabbix	Configuração extra no Grafana
B — CORS no Zabbix	Adicionar Access-Control-Allow-Origin: * no nginx do Zabbix	Simples, sem mudar o Grafana	Precisa editar o nginx do container
C — Dados via JSON estático	Gerar um dados.json a cada 5s via cron/script	Simples, sem CORS	Não é tempo real
Minha recomendação: Opção A (proxy reverso do Grafana). É a mais limpa e não mexe no Zabbix.

🛠️ O que precisamos para a Opção A
Configurar o proxy no grafana.ini:

ini
[proxy]
zabbix_api = http://zabbix-web:8080
Criar um proxy no Grafana: o Grafana tem suporte nativo a proxy reverso via [plugin.grafana-*]? Não exatamente. Precisamos de um plugin de proxy ou de usar o nginx do Grafana.

Alternativa mais simples: usar o Grafana como proxy via datasource Zabbix. O Grafana já faz isso nativamente: o datasource zabbix que você criou usa http://zabbix-web:8080/api_jsonrpc.php (proxy do lado do servidor). O noc3d.html poderia usar a mesma URL via fetch('/api/datasources/proxy/uid/zabbix/...').

Isso é o mais limpo: o Grafana já tem um proxy para o datasource Zabbix. O noc3d.html usa esse proxy, sem CORS.

🎯 Como o proxy do datasource funciona
Quando você configura um datasource no Grafana, ele fica acessível via:

text

No seu caso, o datasource Zabbix tem uid: zabbix (definido no grafana/provisioning/datasources/zabbix.yaml). Então, qualquer requisição para:

É proxied pelo Grafana para o Zabbix (http://zabbix-web:8080/api_jsonrpc.php), sem CORS, porque a origem é a mesma (localhost:3000).

Ótimo ponto. O fato de não ter dado erro de CORS no seu Mac é um ótimo sinal, mas a mudança para Windows 10/11 pode, sim, introduzir diferenças sutis. Vamos analisar o que pode mudar e como mitigar.

🧐 Por que o Mac funcionou e o Windows pode não funcionar?
O CORS (Cross-Origin Resource Sharing) é uma política do navegador, não do sistema operacional. No seu Mac, o navegador (Chrome, Edge, etc.) estava acessando http://localhost:3000 (Grafana) e fazendo requisições para http://localhost:8080 (Zabbix). Embora as portas sejam diferentes, o host é o mesmo (localhost).

Alguns navegadores, em configurações padrão, podem ser mais permissivos com localhost ou o Grafana pode ter conseguido fazer o proxy da requisição de forma transparente. No Windows, o Docker Desktop usa uma rede ligeiramente diferente (WSL2), e o comportamento de rede pode variar, mas o mais provável é que o navegador no Windows seja mais restritivo com a política de mesma origem para portas diferentes.

🛡️ A Solução Definitiva para Produção (Windows)
Para garantir que funcione de forma robusta no Windows, a melhor prática é eliminar a necessidade de CORS, fazendo o Grafana atuar como um proxy reverso para a API do Zabbix. Assim, o noc3d.html fará a requisição para o mesmo domínio e porta do Grafana, e o Grafana se encarrega de encaminhar para o Zabbix.

Você já tentou usar o proxy do datasource do Grafana (/api/datasources/proxy/uid/zabbix/...), mas ele pode ter limitações. A solução mais robusta em produção é configurar um proxy reverso no nível do servidor web que serve o Grafana.

Como você usará Windows, a ferramenta padrão é o IIS (Internet Information Services) com os módulos URL Rewrite e Application Request Routing (ARR). O próprio time do Grafana mantém um tutorial oficial para isso .

🛠️ Passo a Passo para o Windows (Produção)

Baixe e instale o Application Request Routing (ARR) module .

💡 Alternativa: Habilitar CORS no Zabbix (Não Recomendado para Produção)
Se você não quiser mexer no IIS, pode tentar habilitar o CORS diretamente no Zabbix. No entanto, a própria equipe do Zabbix já discutiu isso e considerou uma "ideia realmente ruim do ponto de vista de segurança" permitir que a API aceite requisições de qualquer domínio . Se for fazer isso, restrinja a um domínio específico.

Para o Zabbix 7.0, a configuração de CORS para a API pode ser feita no nginx do container zabbix-web. A forma exata depende da versão da imagem, mas geralmente envolve adicionar cabeçalhos como Access-Control-Allow-Origin no bloco location do nginx .

🚑 Troubleshooting
IIS retorna 404 em /grafana/
Causa: o proxy do ARR não está habilitado.
Solução: IIS Manager → servidor → Application Request Routing Cache → Server Proxy Settings → Enable proxy → Apply.

ERR_TOO_MANY_REDIRECTS
Causa: conflito entre root_url e a regra de rewrite.
Solução: troque {R:2} por {R:0} no web.config.

Grafana mostra "Site can't be reached"
Causa: o IIS não está conseguindo falar com o Grafana.
Solução: verifique se o Grafana está Up (docker compose ps) e se o web.config aponta para http://localhost:3000.

NOC 3D mostra "No data" ou erro de CORS
Causa: o proxy do datasource do Grafana não está configurado.
Solução: verifique se o grafana/provisioning/datasources/zabbix.yaml tem uid: zabbix e se o noc3d.html usa /api/datasources/proxy/uid/zabbix/api_jsonrpc.php.

Certificado SSL não aparece nos bindings do IIS
Causa: o certificado não está no store LocalMachine\My.
Solução: importe o .pfx via Import-PfxCertificate ou use o win-acme para gerar.

Docker Desktop não inicia no Windows
Causa: WSL2 não está configurado.
Solução: wsl --install (PowerShell Admin), reinicie, abra o Docker Desktop.

- **Status:** Aceito
- **Data:** 2026-11-12

## Contexto

O projeto foi desenvolvido no macOS (Mac mini M4). Para produção, é
necessário escolher um sistema operacional.

As opções eram:

1. **Windows 10** — amplamente usado em empresas, mas em fim de suporte
   (outubro de 2025).
2. **Windows 11** — sucessor do Windows 10, suporte até 2031.
3. **Linux** (Ubuntu/Debian) — mais próximo do ambiente Docker, mas
   exige mais conhecimento da equipe.

A equipe de TI da empresa tem **mais familiaridade com Windows** (Active
Directory, GPO, IIS, etc.).

## Decisão

Usar **Windows 10** como ambiente de produção **enquanto o Windows 11 não
for homologado pela equipe de TI**.

**Nota:** o Windows 10 está em fim de suporte (outubro de 2025). A migração
para Windows 11 deve ser planejada.

## Consequências

### Positivas
- **Familiaridade da equipe:** IIS, Active Directory, GPO.
- **Compatibilidade:** Docker Desktop funciona no Windows 10 (via WSL2).
- **Custo:** máquinas já existentes podem ser reutilizadas.

### Negativas
- **Fim de suporte:** o Windows 10 não recebe mais atualizações de
  segurança após outubro de 2025.
- **Risco de segurança:** vulnerabilidades não corrigidas.
- **Necessidade de migração:** planejar a migração para Windows 11.

### Neutras
- O IIS é o mesmo no Windows 10 e 11 (mesmos módulos: URL Rewrite, ARR).
- O Docker Desktop é o mesmo.
- O `web.config` é idêntico.

## Alternativas descartadas

### A) Windows 11
- **O que era:** usar o sucessor do Windows 10.
- **Por que foi descartada:** a equipe de TI ainda não homologou o
  Windows 11 em produção (falta de testes de compatibilidade com
  softwares internos).

### B) Linux
- **O que era:** usar Ubuntu/Debian em produção.
- **Por que foi descartada:** a equipe de TI não tem familiaridade com
  Linux; o Active Directory e o GPO são baseados em Windows.

# O que faz:

#   1. Abra o PowerShell como Administrador

#   3. .\setup-producao.ps1

$ErrorActionPreference = "Stop"

# ============================================================
# CONFIGURAÇÃO (ALTERE AQUI)
# ============================================================
$Dominio = "monitoramento.suaempresa.com"
$SubPath = "grafana"
$GrafanaPorta = 3000
$GrafanaIniPath = "C:\monitoramento-cpd\grafana\config\grafana.ini"

Write-Host "  SETUP DE PRODUÇÃO — monitoramento-cpd" -ForegroundColor Cyan
Write-Host "============================================================" -ForegroundColor Cyan

Write-Host "Sub-path: /$SubPath/"
Write-Host "Porta do Grafana: $GrafanaPorta"
Write-Host ""

# ============================================================
# 1. VERIFICAR SE ESTÁ RODANDO COMO ADMIN
# ============================================================
Write-Host "[1/7] Verificando privilégios..." -ForegroundColor Yellow
$isAdmin = ([Security.Principal.WindowsPrincipal] [Security.Principal.WindowsIdentity]::GetCurrent()).IsInRole([Security.Principal.WindowsBuiltInRole]::Administrator)
if (-not $isAdmin) {
    Write-Host "ERRO: Execute como Administrador." -ForegroundColor Red
    exit 1
}
Write-Host "  OK: Rodando como Administrador." -ForegroundColor Green

# ============================================================
# 2. INSTALAR IIS
# ============================================================
Write-Host "[2/7] Verificando IIS..." -ForegroundColor Yellow
if (-not (Get-WindowsFeature -Name Web-Server).Installed) {
    Write-Host "  Instalando IIS..."
    Enable-WindowsOptionalFeature -Online -FeatureName IIS-WebServerRole -All -NoRestart
    Enable-WindowsOptionalFeature -Online -FeatureName IIS-WebServer -All -NoRestart
    Enable-WindowsOptionalFeature -Online -FeatureName IIS-CommonHttpFeatures -All -NoRestart
    Enable-WindowsOptionalFeature -Online -FeatureName IIS-HttpRedirect -All -NoRestart
    Enable-WindowsOptionalFeature -Online -FeatureName IIS-ApplicationDevelopment -All -NoRestart
    Enable-WindowsOptionalFeature -Online -FeatureName IIS-WebSockets -All -NoRestart
    Enable-WindowsOptionalFeature -Online -FeatureName IIS-ManagementConsole -All -NoRestart
    Write-Host "  OK: IIS instalado." -ForegroundColor Green
} else {
    Write-Host "  OK: IIS já instalado." -ForegroundColor Green

# ============================================================
# 3. INSTALAR URL REWRITE + ARR
# ============================================================
Write-Host "[3/7] Verificando URL Rewrite e ARR..." -ForegroundColor Yellow

# Verifica se o URL Rewrite está instalado
$urlRewritePath = "$env:windir\system32\inetsrv\rewrite.dll"
$urlRewriteInstalled = Test-Path $urlRewritePath

# Verifica se o ARR está instalado
$arrPath = "$env:ProgramFiles\IIS\Application Request Routing"
$arrInstalled = Test-Path $arrPath

if (-not $urlRewriteInstalled -or -not $arrInstalled) {
    Write-Host "  Baixando instaladores (URL Rewrite + ARR)..." -ForegroundColor Yellow

    # URL Rewrite
    if (-not $urlRewriteInstalled) {
        $urlRewriteUrl = "https://download.microsoft.com/download/1/2/8/128E2E22-C1B9-44A4-BE2A-5859ED1D4592/rewrite_amd64_en-US.msi"
        $urlRewriteMsi = "$env:TEMP\rewrite_amd64.msi"
        Write-Host "    Baixando URL Rewrite..."
        Invoke-WebRequest -Uri $urlRewriteUrl -OutFile $urlRewriteMsi -UseBasicParsing
        Write-Host "    Instalando URL Rewrite..."
        Start-Process msiexec.exe -ArgumentList "/i `"$urlRewriteMsi`" /qn /norestart" -Wait
        Write-Host "    OK: URL Rewrite instalado." -ForegroundColor Green
    } else {
        Write-Host "  OK: URL Rewrite já instalado." -ForegroundColor Green
    }

    # ARR
    if (-not $arrInstalled) {
        $arrUrl = "https://download.microsoft.com/download/E/9/8/E9849D6A-020E-47E4-9FD0-A023E99B54EB/requestRouter_amd64.msi"
        $arrMsi = "$env:TEMP\requestRouter_amd64.msi"
        Write-Host "    Baixando ARR..."
        Invoke-WebRequest -Uri $arrUrl -OutFile $arrMsi -UseBasicParsing
        Write-Host "    Instalando ARR..."
        Start-Process msiexec.exe -ArgumentList "/i `"$arrMsi`" /qn /norestart" -Wait
        Write-Host "    OK: ARR instalado." -ForegroundColor Green
    } else {
        Write-Host "  OK: ARR já instalado." -ForegroundColor Green
    }

Import-Module WebAdministration
Set-WebConfigurationProperty -PSPath 'MACHINE/WEBROOT/APPHOST' -Filter "system.webServer/proxy" -Name "enabled" -Value "True"
& $env:windir\system32\inetsrv\appcmd.exe set config -section:system.webServer/proxy -preserveHostHeader:true /commit:apphost

Write-Host "[5/7] Criando web.config..." -ForegroundColor Yellow

# Backup do web.config existente
if (Test-Path $webConfigPath) {
    $backupPath = "$webConfigPath.bak_$(Get-Date -Format 'yyyyMMdd_HHmmss')"
    Copy-Item $webConfigPath $backupPath
    Write-Host "  Backup do web.config existente: $backupPath"

$webConfig = @"

          <action type="Rewrite" url="http://localhost:$GrafanaPorta/{R:2}" logRewrittenUrl="false" />
        </rule>
      </rules>
    </rewrite>
  </system.webServer>
</configuration>
"@

## 🤖 Script de automação (Fase 4)

A **Fase 4 (IIS)** tem um script que automatiza a maior parte do trabalho:

- **Local:** `scripts/setup-producao.ps1`
- **O que ele faz:**
  1. Instala o IIS (se não estiver instalado).
  2. Baixa e instala **URL Rewrite** + **ARR**.
  3. Habilita o proxy no ARR.
  4. Configura `preserveHostHeader` (necessário para Grafana > 8.3.5).
  5. Cria o `web.config` (com backup do existente).
  6. Ajusta o `grafana.ini` (adiciona a seção `[server]`).
  7. Reinicia o IIS.
- **O que ele NÃO faz (manual):** certificado SSL, subir o Docker, testar o NOC 3D.

🎯 Fase 4 — Configurar o IIS (proxy reverso)
Recomendado: use o script scripts/setup-producao.ps1 (automatiza 80% desta fase). Se preferir manual, siga os passos abaixo.

4.1 Instalar IIS
powershell

Enable-WindowsOptionalFeature -Online -FeatureName IIS-HttpRedirect -All
Enable-WindowsOptionalFeature -Online -FeatureName IIS-ApplicationDevelopment -All

Teste:

powershell
start http://localhost
O que esperar: a página padrão do IIS ("Welcome to IIS").

4.2 Instalar URL Rewrite + ARR
Ordem importa: URL Rewrite antes do ARR.

URL Rewrite:

Baixe em https://www.iis.net/downloads/microsoft/url-rewrite.

Instale (.msi).

Reinicie o IIS (iisreset).

ARR:

Baixe em https://www.iis.net/downloads/microsoft/application-request-routing.

Instale (.msi).

Reinicie o IIS (iisreset).

Teste:

Abra o IIS Manager → clique no servidor (não no site) → veja se aparecem os ícones:

URL Rewrite (na seção "IIS")

Application Request Routing Cache (na seção "IIS")

4.3 Habilitar o proxy no ARR
No IIS Manager, clique no servidor (não no site).

Duplo clique em Application Request Routing Cache.

No painel direito, clique em Server Proxy Settings.

Marque Enable proxy.

Apply.

⚠️ Sem isso, o IIS retorna 404 mesmo com o web.config correto.

4.4 Configurar o preserveHostHeader
powershell
# PowerShell como Administrador
& $env:windir\system32\inetsrv\appcmd.exe set config -section:system.webServer/proxy -preserveHostHeader:true /commit:apphost
Por quê: necessário para o Grafana > 8.3.5 funcionar atrás do IIS.

4.5 Criar o web.config
Salve o arquivo em C:\inetpub\wwwroot\web.config:

xml
<?xml version="1.0" encoding="UTF-8"?>
<configuration>
  <system.webServer>
    <rewrite>
      <rules>
        <rule name="Grafana Inbound Rule" enabled="true" stopProcessing="true">
          <match url="grafana(/)?(.*)" />
          <action type="Rewrite" url="http://localhost:3000/{R:2}" logRewrittenUrl="false" />
        </rule>
      </rules>
    </rewrite>
  </system.webServer>
</configuration>
⚠️ Se der ERR_TOO_MANY_REDIRECTS: troque {R:2} por {R:0}:

xml
<action type="Rewrite" url="http://localhost:3000{R:0}" />
4.6 Testar o IIS
powershell
start http://localhost/grafana/
O que esperar: a página de login do Grafana.

Se der 404:

Verifique se o proxy do ARR está habilitado (passo 4.3).

Verifique se o web.config está em C:\inetpub\wwwroot\.

🛠️ Passo 3 — Atualizar o noc3d.html com dados reais
Vou te dar o noc3d.html completo. Ele usa o proxy do datasource do Grafana (/api/datasources/proxy/uid/zabbix/api_jsonrpc.php), sem CORS.

bash
cat > ~/Desktop/monitoramento-cpd/grafana/noc/noc3d.html << 'HTMLEOF'
<!DOCTYPE html>
<html>
<head>
    <meta charset="UTF-8">
    <title>NOC 3D</title>
    <style>
        html, body { margin: 0; padding: 0; width: 100%; height: 100%; overflow: hidden; background: #15181d; }
        #three-container { position: absolute; inset: 0; width: 100%; height: 100%; }
        .mock-badge {
            position: absolute; top: 12px; left: 12px; z-index: 100;
            padding: 6px 12px; border-radius: 3px;
            font: 11px monospace; color: #8e959e;
            background: rgba(17, 18, 23, 0.85);
            border: 1px solid #2c3038;
            pointer-events: none;
        }
        .mock-badge.erro { color: #f2495c; border-color: #f2495c; }
        .mock-badge.ok { color: #73bf69; border-color: #73bf69; }
    </style>
</head>
<body>
    <div id="three-container"></div>
    <div class="mock-badge" id="badge">Conectando ao Zabbix...</div>

    <script type="module">
        import { criarCena } from './cena3d.js';

        const container = document.getElementById('three-container');
        const badge = document.getElementById('badge');
        const cena = criarCena(container);

        // ============================================================
        // CONFIGURAÇÃO
        // ============================================================
        // Usa o proxy do datasource do Grafana (sem CORS).
        const ZABBIX_API = '/api/datasources/proxy/uid/zabbix/api_jsonrpc.php';
        const ZABBIX_USER = 'Admin';
        const ZABBIX_PASSWORD = 'zabbix';
        const HOST_NAME = 'esp32-cpd';
        const INTERVALO_MS = 5000;

        let authToken = null;
        let hostId = null;
        let schemaStatus = null;

        // ============================================================
        // API DO ZABBIX (via proxy do Grafana)
        // ============================================================
        async function zabbixCall(method, params) {
            const body = {
                jsonrpc: '2.0',
                method,
                params,
                id: 1,
            };
            if (authToken) body.auth = authToken;

            const r = await fetch(ZABBIX_API, {
                method: 'POST',
                headers: { 'Content-Type': 'application/json' },
                credentials: 'same-origin',
                body: JSON.stringify(body),
            });
            if (!r.ok) throw new Error(`HTTP ${r.status} em ${method}`);
            const data = await r.json();
            if (data.error) throw new Error(`${method}: ${data.error.message}`);
            return data.result;
        }

        async function login() {
            authToken = await zabbixCall('user.login', {
                username: ZABBIX_USER,
                password: ZABBIX_PASSWORD,
            });
        }

        async function getHost() {
            const hosts = await zabbixCall('host.get', {
                filter: { host: [HOST_NAME] },
            });
            if (!hosts.length) throw new Error(`Host '${HOST_NAME}' não encontrado`);
            hostId = hosts[0].hostid;
        }

        async function getItems() {
            const keys = [
                'esp32.temp', 'esp32.umid', 'esp32.porta', 'esp32.modo',
                'esp32.estado', 'esp32.distancia', 'esp32.invasoes',
                'esp32.ambiente', 'esp32.luz', 'esp32.agua', 'esp32.energia'
            ];
            return await zabbixCall('item.get', {
                hostids: hostId,
                output: ['key_', 'lastvalue'],
                filter: { key_: keys },
            });
        }

        // ============================================================
        // VALIDAÇÃO CONTRA O SCHEMA (opcional)
        // ============================================================
        async function carregarSchema() {
            try {
                const r = await fetch('/public/grafana-custom/firmware/status-schema.json');
                if (r.ok) {
                    schemaStatus = await r.json();
                    console.log('[NOC 3D] Schema carregado.');
                } else {
                    console.warn('[NOC 3D] Schema não disponível, validação desabilitada.');
                }
            } catch (e) {
                console.warn('[NOC 3D] Erro ao carregar schema:', e.message);
            }
        }

        // ============================================================
        // MONTAR DADOS
        // ============================================================
        function montarDados(items) {
            const map = {};
            for (const it of items) map[it.key_] = it.lastvalue;

            const temp = parseFloat(map['esp32.temp']);
            const umid = parseFloat(map['esp32.umid']);
            const porta = map['esp32.porta'] === 'true' || map['esp32.porta'] === '1' ? 1 : 0;
            const agua = map['esp32.agua'] === 'true' || map['esp32.agua'] === '1' ? 1 : 0;
            const energia = map['esp32.energia'] === 'true' || map['esp32.energia'] === '1' ? 1 : 0;

            return {
                ambiente: {
                    online: true,
                    energia: isNaN(energia) ? 1 : energia,
                    agua: isNaN(agua) ? 0 : agua,
                    porta: porta,
                    temperatura: isNaN(temp) ? null : temp,
                    umidade: isNaN(umid) ? null : umid,
                },
                servidores: [
                    { id: 'srv1', nome: 'SRV-01', estado: 'ok', cpu: 30 }
                ],
            };
        }

        // ============================================================
        // LOOP
        // ============================================================
        async function loop() {
            try {
                if (!authToken) {
                    await login();
                    await getHost();
                }
                const items = await getItems();
                const dados = montarDados(items);
                cena.atualizar(dados);

                const porta = dados.ambiente.porta === 1 ? 'porta aberta' : 'porta fechada';
                const temp = dados.ambiente.temperatura != null ? `${dados.ambiente.temperatura.toFixed(1)}°C` : '--';
                badge.textContent = `Zabbix | ${temp} | ${porta}`;
                badge.classList.remove('erro');
                badge.classList.add('ok');
            } catch (e) {
                console.error('[NOC 3D] Erro:', e.message);
                badge.textContent = `Erro: ${e.message}`;
                badge.classList.add('erro');
                badge.classList.remove('ok');
                authToken = null; // força relogin
            }
            setTimeout(loop, INTERVALO_MS);
        }

        // ============================================================
        // INICIAR
        // ============================================================
        (async () => {
            await carregarSchema();
            loop();
        })();
    </script>
</body>
</html>

Quantas mensagens foram perdidas?

Sem #9, o #10 vira "duas salas no escuro". Com #9, o #10 vira "duas salas com métricas de saúde".

#9 também é pré-requisito para o LICENSE (#2)

🚀 Item #9 — Observabilidade do Pipeline (Prometheus + Grafana)
Vamos montar a observabilidade do próprio pipeline. A ideia é adicionar Prometheus para coletar métricas do Mosquitto, do Zabbix e do Docker, e um dashboard no Grafana para visualizar.

🎯 O que vamos construir
Componente	O que faz
Prometheus	Coleta métricas a cada 15s
mosquitto-exporter	Expõe métricas do Mosquitto (mensagens/s, clientes, fila)
cadvisor	Expõe métricas dos containers Docker (CPU, memória, rede)
node-exporter	Expõe métricas do host (CPU, memória, disco)
Grafana (datasource Prometheus)	Provisionado via YAML
Dashboard de observabilidade	Provisionado via JSON
Resultado: um painel com mensagens MQTT/s, CPU/memória dos containers, latência e uptime.

🛠️ Passo 1 — Adicionar os serviços no docker-compose.yml
Abra o docker-compose.yml:

bash
open -e ~/Desktop/monitoramento-cpd/docker-compose.yml
Adicione os 4 serviços abaixo no final do bloco services: (antes de networks:):

yaml
  # ---------------------------------------------------------------
  # Observabilidade — Prometheus + Exporters
  # ---------------------------------------------------------------
  prometheus:
    image: prom/prometheus:v2.54.1
    container_name: cpd-prometheus
    restart: unless-stopped
    ports:
      - "9090:9090"
    volumes:
      - ./prometheus/prometheus.yml:/etc/prometheus/prometheus.yml:ro
      - prometheus_data:/prometheus
    command:
      - '--config.file=/etc/prometheus/prometheus.yml'
      - '--storage.tsdb.path=/prometheus'
      - '--storage.tsdb.retention.time=15d'
      - '--web.enable-lifecycle'
    networks:
      - cpd-net

  mosquitto-exporter:
    image: sapcc/mosquitto-exporter:v0.8.0
    container_name: cpd-mosquitto-exporter
    restart: unless-stopped
    environment:
      BROKER_ENDPOINT: "tcp://cpd-mosquitto:1883"
    ports:
      - "9234:9234"
    depends_on:
      - mosquitto
    networks:
      - cpd-net

  cadvisor:
    image: gcr.io/cadvisor/cadvisor:v0.49.1
    container_name: cpd-cadvisor
    restart: unless-stopped
    ports:
      - "8081:8080"
    volumes:
      - /:/rootfs:ro
      - /var/run:/var/run:ro
      - /sys:/sys:ro
      - /var/lib/docker/:/var/lib/docker:ro
      - /dev/disk/:/dev/disk:ro
    privileged: true
    devices:
      - /dev/kmsg
    networks:
      - cpd-net

  node-exporter:
    image: prom/node-exporter:v1.8.2
    container_name: cpd-node-exporter
    restart: unless-stopped
    ports:
      - "9100:9100"
    command:
      - '--path.procfs=/host/proc'
      - '--path.sysfs=/host/sys'
      - '--collector.filesystem.mount-points-exclude=^/(sys|proc|dev|host|etc)($$|/)'
    volumes:
      - /proc:/host/proc:ro
      - /sys:/host/sys:ro
      - /:/rootfs:ro
    networks:
      - cpd-net
Nota sobre o macOS: o cadvisor e o node-exporter no macOS têm limitações (o Docker Desktop usa uma VM Linux por baixo). As métricas de CPU/memória dos containers funcionam, mas as do host (node-exporter) podem não refletir o Mac. Em produção no Windows, funcionam 100%.
