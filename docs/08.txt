A solução definitiva: selecionar o item pelo Key técnico
O plugin do Zabbix aceita dois modos de filtro:

filter — busca por nome visível (o que estamos usando, e que está falhando).

item — busca pelo Key técnico (ex.: esp32.temp, esp32.modo). Este é mais confiável e não depende do nome visível.

Vamos trocar todos os filtros para usar o Key técnico. Os keys que você já tem no Zabbix são:

🚀 Fatia 5 — Configuração do Zabbix 100% via Código (IaC)
Excelente escolha. Vamos transformar tudo o que você fez na interface do Zabbix (host, itens, templates) em um único script Python que cria tudo do zero, de forma idempotente (pode rodar 100 vezes que não duplica).

Isso é o que separa um laboratório de um ambiente de produção real.

O que vamos construir
Um script configurar_zabbix.py que, a partir do .env, cria automaticamente:

Grupo de hosts (CPD)

Host (esp32-cpd com nome visível Rack CPD)

Item Mestre (mqtt.status, tipo External check)

9 itens dependentes (Temperatura, Umidade, Porta, Modo, Estado, Distância, Invasões, Ambiente, Uptime)

Macros (variáveis configuráveis: {$TEMP.ALERTA}, etc.)

Template exportável (para reutilizar em outra sala)

Nota: Os gatilhos (triggers) entram na Fatia 3 (que você pulou). Por enquanto, o script cria a estrutura de coleta.

Passo 1 — Entender a estrutura do script
O script usará a API do Zabbix (a mesma que usamos no configurar_zabbix.py anterior). A diferença é que agora ele será idempotente: antes de criar, ele verifica se já existe e, se sim, atualiza.

O fluxo será:

text
Login → Verifica Grupo → Cria/Atualiza
     → Verifica Host → Cria/Atualiza
     → Verifica Item Mestre → Cria/Atualiza
     → Para cada item dependente → Verifica → Cria/Atualiza
     → Exporta Template YAML
     → Logout
Passo 2 — Criar o arquivo configurar_zabbix.py

#!/usr/bin/env python3
"""
configurar_zabbix.py — Infraestrutura como Código (IaC) para o Zabbix.

Cria/atualiza toda a estrutura de monitoramento do CPD no Zabbix via API.
Idempotente: pode rodar quantas vezes quiser sem duplicar.

Uso:
    source .venv/bin/activate
    python configurar_zabbix.py
"""

import os
import sys
import requests
from dotenv import load_dotenv

load_dotenv()

ZABBIX_URL = os.getenv("ZABBIX_URL", "http://localhost:8080")
ZABBIX_USER = os.getenv("ZABBIX_USER", "Admin")
ZABBIX_PASSWORD = os.getenv("ZABBIX_PASSWORD", "zabbix")
API_URL = f"{ZABBIX_URL}/api_jsonrpc.php"

# Configurações do ambiente (podem vir do .env)
HOST_NAME = os.getenv("ESP32_HOST_NAME", "esp32-cpd")
HOST_VISIBLE = os.getenv("ESP32_HOST_VISIBLE", "Rack CPD")
HOST_GROUP = os.getenv("ESP32_HOST_GROUP", "CPD")
MQTT_BROKER = os.getenv("MQTT_BROKER_HOST", "cpd-mosquitto")
MQTT_PORT = os.getenv("MQTT_BROKER_PORT", "1883")
MQTT_TOPIC_BASE = os.getenv("MQTT_TOPIC_BASE", "publico/rack_monitor_ABCD1234")

# Item mestre (o script externo que já criamos no Dockerfile)
MASTER_KEY = "mqtt_status.sh"


# ============================================================
# CLIENTE DA API
# ============================================================
class ZabbixAPI:
    def __init__(self, url, user, password):
        self.url = url
        self.user = user
        self.password = password
        self.auth = None
        self.req_id = 0

    def call(self, method, params=None):
        self.req_id += 1
        payload = {
            "jsonrpc": "2.0",
            "method": method,
            "params": params or {},
            "id": self.req_id,
        }
        if self.auth:
            payload["auth"] = self.auth

        r = requests.post(self.url, json=payload, timeout=10)
        r.raise_for_status()
        data = r.json()

        if "error" in data:
            err = data["error"]
            raise RuntimeError(
                f"API {method} falhou: [{err['code']}] {err['message']} — {err.get('data', '')}"
            )
        return data["result"]

    def login(self):
        self.auth = self.call("user.login", {
            "username": self.user,
            "password": self.password,
        })
        return self.auth

    def logout(self):
        if self.auth:
            self.call("user.logout", {})
            self.auth = None


# ============================================================
# HELPERS DE IDEMPOTÊNCIA
# ============================================================
def upsert_group(api, name):
    """Cria ou retorna o grupo de hosts."""
    existing = api.call("hostgroup.get", {"filter": {"name": [name]}})
    if existing:
        print(f"  [=] Grupo '{name}' já existe (ID {existing[0]['groupid']})")
        return existing[0]["groupid"]
    result = api.call("hostgroup.create", {"name": name})
    gid = result["groupids"][0]
    print(f"  [+] Grupo '{name}' criado (ID {gid})")
    return gid


def upsert_host(api, host, visible, group_id):
    """Cria ou atualiza o host."""
    existing = api.call("host.get", {"filter": {"host": [host]}})
    if existing:
        hid = existing[0]["hostid"]
        print(f"  [=] Host '{host}' já existe (ID {hid})")
        # Atualiza nome visível e grupos
        api.call("host.update", {
            "hostid": hid,
            "name": visible,
            "groups": [{"groupid": group_id}],
        })
        return hid

    result = api.call("host.create", {
        "host": host,
        "name": visible,
        "groups": [{"groupid": group_id}],
        "interfaces": [],
        "inventory_mode": -1,
    })
    hid = result["hostids"][0]
    print(f"  [+] Host '{host}' criado (ID {hid})")
    return hid


def upsert_master_item(api, host_id):
    """Cria ou retorna o item mestre (External check)."""
    existing = api.call("item.get", {
        "hostids": host_id,
        "filter": {"key_": [MASTER_KEY]},
    })
    if existing:
        print(f"  [=] Item mestre '{MASTER_KEY}' já existe (ID {existing[0]['itemid']})")
        return existing[0]["itemid"]

    result = api.call("item.create", {
        "hostid": host_id,
        "name": "MQTT Status",
        "key_": MASTER_KEY,
        "type": 21,          # 21 = External check
        "value_type": 4,     # 4 = Text
        "delay": "30s",
        "timeout": "10s",
        "history": "31d",
        "trends": "0",
    })
    iid = result["itemids"][0]
    print(f"  [+] Item mestre criado (ID {iid})")
    return iid


def upsert_dependent_item(api, host_id, master_id, name, key, value_type, units, jsonpath):
    """Cria ou atualiza um item dependente com JSONPath."""
    existing = api.call("item.get", {
        "hostids": host_id,
        "filter": {"key_": [key]},
    })

    payload = {
        "hostid": host_id,
        "name": name,
        "key_": key,
        "type": 18,                  # 18 = Dependent item
        "value_type": value_type,    # 0=float, 3=unsigned, 4=text
        "units": units,
        "master_itemid": master_id,
        "preprocessing": [
            {
                "type": 12,           # 12 = JSONPath
                "params": jsonpath,
                "error_handler": 0,
                "error_handler_params": "",
            }
        ],
        "delay": "0",                # 0 = herda do mestre
        "history": "31d",
        "trends": "365d",
    }

    if existing:
        payload["itemid"] = existing[0]["itemid"]
        api.call("item.update", payload)
        print(f"  [=] Item '{name}' atualizado")
    else:
        api.call("item.create", payload)
        print(f"  [+] Item '{name}' criado")


def upsert_macros(api, host_id):
    """Cria/atualiza macros do host (variáveis configuráveis)."""
    macros = [
        ("{$TEMP.ALERTA}", "28"),
        ("{$TEMP.CRITICO}", "32"),
        ("{$UMID.ALERTA}", "65"),
        ("{$UMID.CRITICO}", "80"),
        ("{$PORTA.TEMPO}", "1m"),
        ("{$ESP32.SEM_DADOS}", "2m"),
    ]
    existing = api.call("usermacro.get", {"hostids": host_id, "output": "extend"})
    existing_names = {m["macro"] for m in existing}

    to_create = [
        {"hostid": host_id, "macro": name, "value": value, "type": 0}
        for name, value in macros
        if name not in existing_names
    ]
    if to_create:
        api.call("usermacro.create", to_create)
        print(f"  [+] {len(to_create)} macros criadas")
    else:
        print(f"  [=] Macros já existem")


def export_template(api, host_id):
    """Exporta o template YAML com todos os itens do host."""
    try:
        result = api.call("configuration.export", {
            "format": "yaml",
            "options": {"hosts": [host_id]},
        })
        with open("zabbix/template_esp32_cpd.yaml", "w") as f:
            f.write(result)
        print(f"  [+] Template exportado para zabbix/template_esp32_cpd.yaml")
    except Exception as e:
        print(f"  [!] Aviso: não foi possível exportar template: {e}")


# ============================================================
# MAIN
# ============================================================
def main():
    print("=" * 60)
    print("  CONFIGURAÇÃO DO ZABBIX VIA IaC")
    print("=" * 60)
    print(f"Servidor: {ZABBIX_URL}")
    print(f"Host: {HOST_NAME} / {HOST_VISIBLE}")
    print(f"Grupo: {HOST_GROUP}")
    print(f"Broker MQTT: {MQTT_BROKER}:{MQTT_PORT}")
    print()

    api = ZabbixAPI(API_URL, ZABBIX_USER, ZABBIX_PASSWORD)

    try:
        print("[1/6] Autenticando...")
        api.login()
        print("  [+] Login OK")

        print(f"[2/6] Grupo '{HOST_GROUP}'...")
        group_id = upsert_group(api, HOST_GROUP)

        print(f"[3/6] Host '{HOST_NAME}'...")
        host_id = upsert_host(api, HOST_NAME, HOST_VISIBLE, group_id)

        print("[4/6] Item mestre...")
        master_id = upsert_master_item(api, host_id)

        print("[5/6] Itens dependentes...")
        items = [
            # (nome, key, value_type, units, jsonpath)
            ("Temperatura",    "esp32.temp",       0, "°C",  "$.temp_c"),
            ("Umidade",        "esp32.umid",       0, "%",   "$.umid_pct"),

def upsert_dependent_item(api, host_id, master_id, name, key, value_type, units, jsonpath):
    """Cria ou atualiza um item dependente com JSONPath."""
    existing = api.call("item.get", {
        "hostids": host_id,
        "filter": {"key_": [key]},
    })

    # Campos que SEMPRE podem ser atualizados
    payload_update = {
        "name": name,
        "key_": key,
        "type": 18,                  # 18 = Dependent item
        "value_type": value_type,    # 0=float, 3=unsigned, 4=text
        "units": units,
        "preprocessing": [
            {
                "type": 12,           # 12 = JSONPath
                "params": jsonpath,
                "error_handler": 0,
                "error_handler_params": "",
            }
        ],
        "delay": "0",
        "history": "31d",
        "trends": "365d",
    }

    if existing:
        # UPDATE: adiciona só o itemid, NÃO o hostid nem master_itemid
        payload_update["itemid"] = existing[0]["itemid"]
        api.call("item.update", payload_update)
        print(f"  [=] Item '{name}' atualizado")
    else:
        # CREATE: inclui hostid e master_itemid
        payload_create = dict(payload_update)
        payload_create["hostid"] = host_id
        payload_create["master_itemid"] = master_id
        api.call("item.create", payload_create)
        print(f"  [+] Item '{name}' criado")
