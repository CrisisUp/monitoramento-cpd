class ZabbixAPI:
    def __init__(self, url, user, password):
        self.url = url
        self.user = user
        self.password = password
        self.auth = None
        self.req_id = 0

    def call(self, method, params=None):
        self.req_id += 1
        payload = {
            "jsonrpc": "2.0",
            "method": method,
            "params": params or {},
            "id": self.req_id,
        }
        if self.auth:
            payload["auth"] = self.auth

        r = requests.post(self.url, json=payload, timeout=15)
        r.raise_for_status()
        data = r.json()

        if "error" in data:
            err = data["error"]
            raise RuntimeError(
                f"API {method} falhou: [{err['code']}] {err['message']} — {err.get('data', '')}"
            )
        return data["result"]

    def login(self):
        self.auth = self.call("user.login", {
            "username": self.user,
            "password": self.password,
        })
        return self.auth

    def logout(self):
        if self.auth:
            try:
                self.call("user.logout", {})
            except Exception:
                pass
            self.auth = None

def upsert_group(api, name):
    """Cria ou retorna o grupo de hosts."""
    existing = api.call("hostgroup.get", {"filter": {"name": [name]}})
    if existing:
        print(f"  [=] Grupo '{name}' já existe (ID {existing[0]['groupid']})")
        return existing[0]["groupid"]
    result = api.call("hostgroup.create", {"name": name})
    gid = result["groupids"][0]
    print(f"  [+] Grupo '{name}' criado (ID {gid})")
    return gid


def upsert_host(api, host, visible, group_id):
    """Cria ou atualiza o host."""
    existing = api.call("host.get", {"filter": {"host": [host]}})
    if existing:
        hid = existing[0]["hostid"]
        print(f"  [=] Host '{host}' já existe (ID {hid})")
        api.call("host.update", {
            "hostid": hid,
            "name": visible,
            "groups": [{"groupid": group_id}],
        })
        return hid

    result = api.call("host.create", {
        "host": host,
        "name": visible,
        "groups": [{"groupid": group_id}],
        "interfaces": [],
        "inventory_mode": -1,
    })
    hid = result["hostids"][0]
    print(f"  [+] Host '{host}' criado (ID {hid})")
    return hid


def upsert_master_item(api, host_id):
    """Cria ou retorna o item mestre (External check)."""
    existing = api.call("item.get", {
        "hostids": host_id,
        "filter": {"key_": [MASTER_KEY]},
    })

    payload_update = {
        "name": "MQTT Status",
        "key_": MASTER_KEY,
        "type": 21,          # 21 = External check
        "value_type": 4,     # 4 = Text
        "delay": "30s",
        "history": "31d",
        "trends": "0",
    }

    if existing:
        itemid = existing[0]["itemid"]
        payload_update["itemid"] = itemid
        api.call("item.update", payload_update)
        print(f"  [=] Item mestre '{MASTER_KEY}' atualizado (ID {itemid})")
        return itemid

    payload_create = dict(payload_update)
    payload_create["hostid"] = host_id
    result = api.call("item.create", payload_create)
    itemid = result["itemids"][0]
    print(f"  [+] Item mestre '{MASTER_KEY}' criado (ID {itemid})")
    return itemid


def upsert_dependent_item(api, host_id, master_id, name, key, value_type, units, jsonpath):
    """Cria ou atualiza um item dependente com JSONPath."""
    existing = api.call("item.get", {
        "hostids": host_id,
        "filter": {"key_": [key]},
    })

    # Campos comuns (update + create)
    payload_update = {
        "name": name,
        "key_": key,
        "type": 18,                  # 18 = Dependent item
        "value_type": value_type,    # 0=float, 3=unsigned, 4=text
        "preprocessing": [
            {
                "type": 12,           # 12 = JSONPath
                "params": jsonpath,
                "error_handler": 0,
                "error_handler_params": "",
            }
        ],
        "delay": "0",                # 0 = herda do mestre
        "history": "31d",
        "trends": "365d",
    }

    # Adiciona units só se não for vazio (alguns tipos rejeitam "")
    if units:
        payload_update["units"] = units

    if existing:
        # UPDATE: só itemid (hostid e master_itemid são imutáveis)
        payload_update["itemid"] = existing[0]["itemid"]
        api.call("item.update", payload_update)
        print(f"  [=] Item '{name}' atualizado")
    else:
        # CREATE: inclui hostid e master_itemid
        payload_create = dict(payload_update)
        payload_create["hostid"] = host_id
        payload_create["master_itemid"] = master_id
        api.call("item.create", payload_create)
        print(f"  [+] Item '{name}' criado")


def upsert_macros(api, host_id):
    """Cria/atualiza macros do host (variáveis configuráveis)."""
    macros = [
        ("{$TEMP.ALERTA}", "28"),
        ("{$TEMP.CRITICO}", "32"),
        ("{$UMID.ALERTA}", "65"),
        ("{$UMID.CRITICO}", "80"),
        ("{$PORTA.TEMPO}", "1m"),
        ("{$ESP32.SEM_DADOS}", "2m"),
    ]
    existing = api.call("usermacro.get", {"hostids": host_id, "output": "extend"})
    existing_names = {m["macro"] for m in existing}

    to_create = [
        {"hostid": host_id, "macro": name, "value": value, "type": 0}
        for name, value in macros
        if name not in existing_names
    ]
    if to_create:
        api.call("usermacro.create", to_create)
        print(f"  [+] {len(to_create)} macro(s) criada(s)")
    else:
        print(f"  [=] Macros já existem")


def export_template(api, host_id):
    """Exporta o template YAML com todos os itens do host."""
    try:
        # Garante que a pasta zabbix/ existe
        os.makedirs("zabbix", exist_ok=True)
        result = api.call("configuration.export", {
            "format": "yaml",
            "options": {"hosts": [host_id]},
        })
        path = "zabbix/template_esp32_cpd.yaml"
        with open(path, "w") as f:
            f.write(result)
        print(f"  [+] Template exportado para {path}")
    except Exception as e:
        print(f"  [!] Aviso: não foi possível exportar template: {e}")

class ZabbixAPI:
    def __init__(self, url, user, password):
        self.url = url
        self.user = user
        self.password = password
        self.auth = None
        self.req_id = 0

    def call(self, method, params=None):
        self.req_id += 1
        payload = {
            "jsonrpc": "2.0",
            "method": method,
            "params": params or {},
            "id": self.req_id,
        }
        if self.auth:
            payload["auth"] = self.auth

        r = requests.post(self.url, json=payload, timeout=15)
        r.raise_for_status()
        data = r.json()

        if "error" in data:
            err = data["error"]
            raise RuntimeError(
                f"API {method} falhou: [{err['code']}] {err['message']} — {err.get('data', '')}"
            )
        return data["result"]

    def login(self):
        self.auth = self.call("user.login", {
            "username": self.user,
            "password": self.password,
        })
        return self.auth

    def logout(self):
        if self.auth:
            try:
                self.call("user.logout", {})
            except Exception:
                pass
            self.auth = None

🎯 O que essa fatia te ensinou
A API do Zabbix é a "porta dos fundos" — ela expõe tudo o que a interface faz, mas de forma scriptável.

Idempotência não é "só rodar de novo" — é detectar o estado antes de agir. O padrão upsert (get → create ou update) é o coração do IaC.

Regras específicas da API do Zabbix 7.0:

trends só pode ser > 0 em itens float (não em texto/unsigned).

hostid e master_itemid são imutáveis após a criação (não vão no update).

preprocessing vazio é rejeitado no update.

Macros são variáveis — mudar {$TEMP.ALERTA} de 28 para 30 é editar o .env e rodar o script, em vez de clicar 4 vezes na interface.

O template exportado é o "contrato" — você pode importá-lo em outro Zabbix e ter os mesmos itens.

### Fatia 5 — IaC do Zabbix

- **Zabbix 7.0 rejeita `trends > 0` em itens não-float.**
  Para texto (`value_type=4`) e unsigned (`value_type=3`), `trends` deve ser `"0"`.
  Só `float` (`value_type=0`) aceita `trends: "365d"`.

- **`hostid` e `master_itemid` são imutáveis no `item.update`.**
  Você define uma vez no `item.create` e não pode mudar depois. No update, omitir.

- **`preprocessing` vazio é rejeitado no `item.update`.**
  Se o item não tem preprocessor, simplesmente não envie o campo — não envie `[]`.

- **Separar payload de create vs update é mais seguro que reutilizar.**
  Padrão usado no script: `payload = {...campos_mutaveis...}`, depois
  `payload["itemid"] = id` (para update) OU `payload["hostid"] = host_id` (para create).

- **Exportar o template YAML** cria uma "foto" do estado atual, útil para
  migrar para outro Zabbix ou versionar no Git.
