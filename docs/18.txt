#!/usr/bin/env python3
"""Corrige o cpd-dashboard.json com os campos de query preenchidos."""
import json
import os

BASE = "/Users/cristiano/Desktop/monitoramento-cpd"
CAMINHO = os.path.join(BASE, "grafana", "dashboards", "cpd-dashboard.json")

with open(CAMINHO) as f:
    dashboard = json.load(f)

# Configuração padrão para os painéis do Zabbix
GROUP = "CPD"
HOST = "Rack CPD"

# Mapeamento: título do painel -> item do Zabbix
ITENS = {
    "Temperatura": "Temperatura",
    "Umidade": "Umidade",
    "Modo": "Modo",
    "Estado": "Estado",
    "Histórico": None,  # tratado separado (2 queries)
}

for panel in dashboard["panels"]:
    titulo = panel.get("title", "")

    if titulo in ["Temperatura", "Umidade", "Modo", "Estado"]:
        item = ITENS[titulo]
        for target in panel["targets"]:
            target["queryType"] = "0"
            target["group"] = GROUP
            target["host"] = HOST
            target["item"] = item
            target["itemTag"] = ""
            target["itemTagValue"] = ""

    if titulo == "Histórico (30 min)":
        # Painel com 2 targets: Temperatura e Umidade
        itens_historico = ["Temperatura", "Umidade"]
        for i, target in enumerate(panel["targets"]):
            if i < len(itens_historico):
                target["queryType"] = "0"
                target["group"] = GROUP
                target["host"] = HOST
                target["item"] = itens_historico[i]
                target["itemTag"] = ""
                target["itemTagValue"] = ""

with open(CAMINHO, "w") as f:
    json.dump(dashboard, f, indent=2, ensure_ascii=False)

print(f"OK: {CAMINHO}")
print("Painéis corrigidos:")
for panel in dashboard["panels"]:
    titulo = panel.get("title", "")
    if titulo:
        targets_info = []
        for t in panel.get("targets", []):
            if "item" in t:
                targets_info.append(f"{t.get('item', '?')}")
        if targets_info:
            print(f"  - {titulo}: {', '.join(targets_info)}")

import json
import os
import requests
from dotenv import load_dotenv

load_dotenv()

GRAFANA_URL = "http://localhost:3000"

# ============================================================

def painel_stat(titulo, item, unidade, cor_limiar=None):

            "textMode": "auto",
        },
        "targets": [
            {
                "datasource": {"type": "alexanderzobnin-zabbix-datasource", "uid": "zabbix"},
                "queryType": "0",
                "refId": "A",
                "group": "CPD",

    "title": "CPD Dashboard 2D",
    "uid": "cpd-dashboard-2d",
    "version": 1,

# ============================================================
# CRIAR VIA API

}

#!/usr/bin/env python3
"""Corrige o cpd-dashboard.json trocando 'key' por 'filter'."""
import json
import os

BASE = "/Users/cristiano/Desktop/monitoramento-cpd"
CAMINHO = os.path.join(BASE, "grafana", "dashboards", "cpd-dashboard.json")

with open(CAMINHO) as f:
    dashboard = json.load(f)

# Mapeamento: key técnica -> nome do item no Zabbix
MAP = {
    "esp32.umid": "Umidade",
    "esp32.modo": "Modo",
    "esp32.estado": "Estado",
    "esp32.temp": "Temperatura",
    "esp32.porta": "Porta Aberta",
    "esp32.distancia": "Distância",
    "esp32.invasoes": "Invasões",
    "esp32.ambiente": "Ambiente",
    "esp32.luz": "Luminosidade",
}

def corrigir_no(obj):
    """Percorre o JSON e troca 'key' por 'filter'."""
    if isinstance(obj, dict):
        if "item" in obj and isinstance(obj["item"], dict):
            item = obj["item"]
            if "key" in item and item["key"] in MAP:
                item["filter"] = MAP[item["key"]]
                del item["key"]
        for v in obj.values():
            corrigir_no(v)
    elif isinstance(obj, list):
        for v in obj:
            corrigir_no(v)

corrigir_no(dashboard)

with open(CAMINHO, "w") as f:
    json.dump(dashboard, f, indent=2, ensure_ascii=False)

print(f"OK: {CAMINHO}")
print("Itens corrigidos:")
for key, name in MAP.items():
    print(f"  {key} -> {name}")

#!/usr/bin/env python3
"""Corrige o cpd-dashboard.json (Modo e Estado)."""
import json
import os

BASE = "/Users/cristiano/Desktop/monitoramento-cpd"
CAMINHO = os.path.join(BASE, "grafana", "dashboards", "cpd-dashboard.json")

with open(CAMINHO) as f:
    dashboard = json.load(f)

for elem in dashboard["spec"]["elements"].values():
    if elem.get("kind") != "Panel":
        continue
    spec = elem["spec"]
    titulo = spec.get("title", "")
    if titulo not in ["Modo", "Estado"]:
        continue

    # Corrigir queryType
    for q in spec["data"]["spec"]["queries"]:
        if "query" in q["spec"] and "spec" in q["spec"]["query"]:
            qspec = q["spec"]["query"]["spec"]
            qspec["queryType"] = "0"

    # Garantir que o vizConfig.group seja "table"
    spec["vizConfig"]["group"] = "table"
    spec["vizConfig"]["spec"]["options"] = {
        "cellHeight": "lg",
        "displayMode": "gradient",
        "showHeader": False,
    }
    # Adicionar transformação Organize se não existir
    transfs = spec["data"]["spec"].get("transformations", [])
    if not any(t.get("group") == "organize" for t in transfs):
        transfs.append({
            "kind": "Transformation",
            "group": "organize",
            "spec": {"options": {"excludeByName": {"Time": True}}}
        })

with open(CAMINHO, "w") as f:
    json.dump(dashboard, f, indent=2, ensure_ascii=False)

print("OK: Modo e Estado corrigidos (queryType: 0, vizConfig: table)")

O painel table (usado para Modo e Estado) tem um campo options onde definimos o tipo de gráfico. Para o painel stat, o campo "Fields" (que estava em Numeric Fields) pode ser mudado para All Fields via JSON.

#!/usr/bin/env python3
"""Corrige os painéis Modo e Estado: queryType=2 (Text), tipo stat, Fields=All."""
import json
import os
import subprocess

BASE = "/Users/cristiano/Desktop/monitoramento-cpd"
CAMINHO = os.path.join(BASE, "grafana", "dashboards", "cpd-dashboard.json")

# 1. Parar o Grafana
print("[1/4] Parando o Grafana...")
subprocess.run(["docker", "compose", "stop", "grafana"], cwd=BASE, check=True)

# 2. Editar o JSON
print("[2/4] Editando o JSON...")
with open(CAMINHO) as f:
    dashboard = json.load(f)

for panel in dashboard.get("panels", []):
    titulo = panel.get("title", "")
    if titulo not in ["Modo", "Estado"]:
        continue

    # Trocar queryType para "2" (Text)
    for target in panel.get("targets", []):
        target["queryType"] = "2"
        print(f"  {titulo}: queryType = 2 (Text)")

    # Mudar o tipo de gráfico para "stat"
    panel["type"] = "stat"
    print(f"  {titulo}: tipo = stat")

    # Mudar o Fields para "All Fields" (para aceitar texto)
    # No painel stat, o campo é "options.reduceOptions.fields"
    if "options" not in panel:
        panel["options"] = {}
    if "reduceOptions" not in panel["options"]:
        panel["options"]["reduceOptions"] = {}
    panel["options"]["reduceOptions"]["fields"] = "/.*/"
    print(f"  {titulo}: Fields = All Fields (/.*/)")

    # Garantir que o fieldConfig tenha os mappings corretos
    if "fieldConfig" not in panel:
        panel["fieldConfig"] = {"defaults": {}, "overrides": []}
    if "defaults" not in panel["fieldConfig"]:
        panel["fieldConfig"]["defaults"] = {}
    panel["fieldConfig"]["defaults"]["mappings"] = [
        {"type": "value", "options": {"ARMADO": {"color": "blue", "index": 0}, "DESARMADO": {"color": "orange", "index": 1}}} if titulo == "Modo" else
        {"type": "value", "options": {
            "NORMAL": {"color": "green", "index": 0},
            "AVISO SALA": {"color": "yellow", "index": 1},
            "APROXIMACAO RACK": {"color": "orange", "index": 2},
            "EMERGENCIA RACK": {"color": "red", "index": 3},
            "INVASAO SALA": {"color": "red", "index": 4},
            "ACESSO RFID": {"color": "purple", "index": 5},
        }}
    ]

    # Remover transformações que não são necessárias (organize)
    if "transformations" in panel:
        panel["transformations"] = []

with open(CAMINHO, "w") as f:
    json.dump(dashboard, f, indent=2, ensure_ascii=False)
print("  OK: arquivo salvo")

# 3. Subir o Grafana
print("[3/4] Subindo o Grafana...")
subprocess.run(["docker", "compose", "start", "grafana"], cwd=BASE, check=True)

# 4. Aguardar e exportar do banco
print("[4/4] Aguardando 10s e exportando do banco...")
import time
time.sleep(10)

import requests
r = requests.get("http://localhost:3000/api/dashboards/uid/cpd-dashboard", auth=("admin", "admin"))
if r.status_code == 200:
    db_dashboard = r.json()["dashboard"]
    with open(CAMINHO, "w") as f:
        json.dump(db_dashboard, f, indent=2, ensure_ascii=False)
    print("  OK: dashboard exportado do banco para o disco")
else:
    print(f"  AVISO: não foi possível exportar ({r.status_code})")

print("\n✅ Concluído!")
print("Abra: http://localhost:3000/d/cpd-dashboard/cpd-dashboard")
