Por que o filtro por key falhou
Quando o plugin do Zabbix busca um item por "item": { "key": "esp32.modo" }, ele precisa que o datasource tenha acesso ao key técnico. Mas o plugin, por padrão, filtra por nome visível (name), não por key.

A forma correta de filtrar por key técnico no plugin alexanderzobnin-zabbix-app é usar "item": { "key": "esp32.modo" } E "item": { "filter": "Modo" } juntos, ou usar o nome exato do item no Zabbix.

Como no Zabbix o item se chama "MQTT Status: Modo" (porque o Zabbix prefixa dependentes com o nome do mestre), o filtro correto é pelo nome exato.

import os, requests, json
from dotenv import load_dotenv
load_dotenv()

url = "http://localhost:8080/api_jsonrpc.php"
auth = requests.post(url, json={
    "jsonrpc": "2.0", "method": "user.login", "id": 1,
    "params": {"username": "Admin", "password": "zabbix"}
}).json()["result"]

host_id = requests.post(url, json={
    "jsonrpc": "2.0", "method": "host.get", "id": 1, "auth": auth,
    "params": {"filter": {"host": ["esp32-cpd"]}}
}).json()["result"][0]["hostid"]

items = requests.post(url, json={
    "jsonrpc": "2.0", "method": "item.get", "id": 1, "auth": auth,
    "params": {"hostids": host_id, "output": ["itemid", "name", "key_"]}
}).json()["result"]

print(f"{'NAME (nome visível)':<40} {'KEY (técnico)'}")
print("-" * 80)
for it in sorted(items, key=lambda x: x["name"]):
    print(f"{it['name']:<40} {it['key_']}")

PASS = os.getenv("ZABBIX_PASSWORD", "zabbix")
HOST = os.getenv("ESP32_HOST_NAME", "esp32-cpd")


def api(method, params, auth=None):
    payload = {"jsonrpc": "2.0", "method": method, "params": params, "id": 1}
    if auth:
        payload["auth"] = auth
    r = requests.post(URL, json=payload, timeout=10).json()
    if "error" in r:
        raise RuntimeError(r["error"])
    return r["result"]


def main():
    print(f"Conectando em {URL} ...")
    auth = api("user.login", {"username": USER, "password": PASS})
    print(f"  Login OK\n")

    hosts = api("host.get", {"filter": {"host": [HOST]}}, auth)
    if not hosts:
        print(f"❌ Host '{HOST}' não encontrado.")
        return
    host_id = hosts[0]["hostid"]
    print(f"Host '{HOST}' → ID {host_id}\n")

    items = api("item.get", {
        "hostids": host_id,
        "output": ["itemid", "name", "key_", "value_type", "status", "lastvalue", "lastclock"],
        "sortfield": "name",
    }, auth)

    print(f"{'NAME':<28} {'KEY':<20} {'VT':<3} {'ST':<4} {'LAST CLOCK':<12} {'LAST VALUE'}")
    print("-" * 95)
    for it in items:
        vt = it["value_type"]
        st = "ON" if it["status"] == "0" else "OFF"
        last = (it.get("lastvalue") or "(vazio)")[:30]
        clock = it.get("lastclock", "0")
        print(f"{it['name']:<28} {it['key_']:<20} {vt:<3} {st:<4} {clock:<12} {last}")

    print()
    print("Legenda: VT = value_type (0=float, 3=unsigned, 4=text)")
    print("         ST = status (ON=habilitado, OFF=desabilitado)")
    print()
    print(f"Total de itens: {len(items)}")

    api("user.logout", {}, auth)


if __name__ == "__main__":
    main()

O Grafana relê o JSON e recria o painel com "table".

Por que table funciona e stat não
stat é um painel de valor único e foi feito para números. Quando recebe texto, ele tenta converter e falha.

table aceita qualquer tipo de dado. Ele mostra o texto literal na célula.

Como os itens "Modo" e "Estado" são texto (value_type=4), table é o painel correto.
