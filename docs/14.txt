#!/usr/bin/env python3
"""
configurar_telegram.py — Configura o Telegram no Zabbix via API.

Cria/atualiza:
- Tipo de mídia Telegram (com o token do bot).
- Usuário Admin com o Chat ID.
- Triggers para temperatura, porta, água, energia.
- Ação que envia alertas via Telegram.

Uso:
    source .venv/bin/activate
    python configurar_telegram.py
"""

import os
import sys
import requests
from dotenv import load_dotenv

load_dotenv()

ZABBIX_URL = os.getenv("ZABBIX_URL", "http://localhost:8080")
ZABBIX_USER = os.getenv("ZABBIX_USER", "Admin")
ZABBIX_PASSWORD = os.getenv("ZABBIX_PASSWORD", "zabbix")
API_URL = f"{ZABBIX_URL}/api_jsonrpc.php"

TELEGRAM_TOKEN = os.getenv("TELEGRAM_BOT_TOKEN", "")
TELEGRAM_CHAT_ID = os.getenv("TELEGRAM_CHAT_ID", "")
HOST_NAME = os.getenv("ESP32_HOST_NAME", "esp32-cpd")
HOST_VISIBLE = os.getenv("ESP32_HOST_VISIBLE", "Rack CPD")


class ZabbixAPI:
    def __init__(self, url, user, password):
        self.url = url
        self.user = user
        self.password = password
        self.auth = None
        self.req_id = 0

    def call(self, method, params=None):
        self.req_id += 1
        payload = {
            "jsonrpc": "2.0",
            "method": method,
            "params": params or {},
            "id": self.req_id,
        }
        if self.auth:
            payload["auth"] = self.auth
        r = requests.post(self.url, json=payload, timeout=15)
        r.raise_for_status()
        data = r.json()
        if "error" in data:
            err = data["error"]
            raise RuntimeError(f"API {method}: [{err['code']}] {err['message']} — {err.get('data', '')}")
        return data["result"]

    def login(self):
        self.auth = self.call("user.login", {"username": self.user, "password": self.password})
        return self.auth

    def logout(self):
        if self.auth:
            try:
                self.call("user.logout", {})
            except Exception:
                pass
            self.auth = None


def upsert_media_type(api):
    """Cria/atualiza o tipo de mídia Telegram."""
    existing = api.call("mediatype.get", {"filter": {"name": ["Telegram"]}})
    if existing:
        print(f"  [=] Tipo de mídia 'Telegram' já existe (ID {existing[0]['mediatypeid']})")
        return existing[0]["mediatypeid"]

    result = api.call("mediatype.create", {
        "name": "Telegram",
        "type": 4,  # 4 = Webhook
        "exec_path": "",
        "gsm_modem": "",
        "smtp_server": "",
        "smtp_helo": "",
        "smtp_email": "",
        "smtp_security": 0,
        "smtp_verify_host": 0,
        "smtp_verify_peer": 0,
        "smtp_username": "",
        "smtp_password": "",
        "smtp_port": 0,
        "passwd": "",
        "params": [
            {"name": "URL", "value": f"https://api.telegram.org/bot{TELEGRAM_TOKEN}/sendMessage"},
            {"name": "To", "value": TELEGRAM_CHAT_ID},
            {"name": "Message", "value": "{ALERT.MESSAGE}"},
        ],
        "webhook_url": "",
        "status": 0,
        "maxsessions": 1,
        "maxattempts": 3,
        "attempt_interval": "10s",
    })
    mid = result["mediatypeids"][0]
    print(f"  [+] Tipo de mídia 'Telegram' criado (ID {mid})")
    return mid


def upsert_user_media(api, media_type_id):
    """Associa o Chat ID ao usuário Admin."""
    users = api.call("user.get", {"filter": {"username": ["Admin"]}})
    if not users:
        print("  [!] Usuário Admin não encontrado")
        return
    user_id = users[0]["userid"]
    existing = api.call("user.get", {"userids": user_id, "selectMedias": "extend"})
    has_telegram = any(m["mediatypeid"] == media_type_id for m in existing[0].get("medias", []))
    if has_telegram:
        print(f"  [=] Admin já tem Telegram configurado")
        return
    api.call("user.update", {
        "userid": user_id,
        "medias": [{
            "mediatypeid": media_type_id,
            "sendto": TELEGRAM_CHAT_ID,
            "active": 0,
            "severity": 63,  # todas as severidades
            "period": "1-7,00:00-24:00",
        }],
    })
    print(f"  [+] Telegram associado ao Admin (Chat ID: {TELEGRAM_CHAT_ID})")


def upsert_triggers(api, host_id):
    """Cria/atualiza os triggers para os itens dependentes."""
    triggers = [
        {
            "description": "Temperatura acima de 30°C",
            "expression": f"last(/{HOST_NAME}/esp32.temp)>=30",
            "priority": 3,  # 3 = Average
            "tags": [{"tag": "efeito", "value": "temperatura"}],
        },
        {
            "description": "Temperatura crítica acima de 35°C",
            "expression": f"last(/{HOST_NAME}/esp32.temp)>=35",
            "priority": 5,  # 5 = Disaster
            "tags": [{"tag": "efeito", "value": "temperatura"}],
        },
        {
            "description": "Porta aberta",
            "expression": f"last(/{HOST_NAME}/esp32.porta)=\"true\"",
            "priority": 4,  # 4 = High
            "tags": [{"tag": "efeito", "value": "porta"}],
        },
        {
            "description": "Água detectada no piso",
            "expression": f"last(/{HOST_NAME}/esp32.agua)=1",
            "priority": 5,
            "tags": [{"tag": "efeito", "value": "agua"}],
        },
        {
            "description": "Tomada sem energia",
            "expression": f"last(/{HOST_NAME}/esp32.energia)=0",
            "priority": 4,
            "tags": [{"tag": "efeito", "value": "energia"}],
        },
    ]

    for t in triggers:
        existing = api.call("trigger.get", {
            "hostids": host_id,
            "filter": {"description": [t["description"]]},
        })
        if existing:
            print(f"  [=] Trigger '{t['description']}' já existe")
            continue
        api.call("trigger.create", {
            "description": t["description"],
            "expression": t["expression"],
            "priority": t["priority"],
            "tags": t.get("tags", []),
        })
        print(f"  [+] Trigger '{t['description']}' criado")


def upsert_action(api, media_type_id):
    """Cria/atualiza a ação que envia alertas via Telegram."""
    existing = api.call("action.get", {"filter": {"name": ["Sala do servidor -> Telegram"]}})
    if existing:
        print(f"  [=] Ação 'Sala do servidor -> Telegram' já existe")
        return

    api.call("action.create", {
        "name": "Sala do servidor -> Telegram",
        "eventsource": 0,  # 0 = triggers
        "status": 0,
        "esc_period": "1h",
        "filter": {
            "evaltype": 0,
            "conditions": [
                {"conditiontype": 0, "operator": 0, "value": "", "formulaid": "A"},
            ],
        },
        "operations": [
            {
                "operationtype": 0,
                "esc_step_from": 1,
                "esc_step_to": 1,
                "evaltype": 0,
                "opmessage": {
                    "default_msg": 1,
                    "mediatypeid": media_type_id,
                },
                "opmessage_grp": [],
                "opmessage_usr": [{"userid": "1"}],  # Admin
            },
        ],
        "recovery_operations": [
            {
                "operationtype": 0,
                "opmessage": {
                    "default_msg": 1,
                    "mediatypeid": media_type_id,
                },
                "opmessage_usr": [{"userid": "1"}],
            },
        ],
    })
    print(f"  [+] Ação 'Sala do servidor -> Telegram' criada")


def main():
    print("=" * 60)
    print("  CONFIGURAÇÃO DO TELEGRAM VIA IaC")
    print("=" * 60)

    if not TELEGRAM_TOKEN or not TELEGRAM_CHAT_ID:
        print("❌ ERRO: Configure TELEGRAM_BOT_TOKEN e TELEGRAM_CHAT_ID no .env")
        sys.exit(1)

    api = ZabbixAPI(API_URL, ZABBIX_USER, ZABBIX_PASSWORD)

    try:
        print("[1/5] Autenticando...")
        api.login()
        print("  [+] Login OK")

        print("[2/5] Buscando host...")
        hosts = api.call("host.get", {"filter": {"host": [HOST_NAME]}})
        if not hosts:
            print(f"❌ Host '{HOST_NAME}' não encontrado.")
            sys.exit(1)
        host_id = hosts[0]["hostid"]
        print(f"  [+] Host '{HOST_NAME}' encontrado (ID {host_id})")

        print("[3/5] Configurando tipo de mídia Telegram...")
        media_type_id = upsert_media_type(api)

        print("[4/5] Associando Chat ID ao Admin...")
        upsert_user_media(api, media_type_id)

        print("[5/5] Criando triggers e ação...")
        upsert_triggers(api, host_id)
        upsert_action(api, media_type_id)

        print()
        print("=" * 60)
        print("  ✅ CONFIGURAÇÃO CONCLUÍDA")
        print("=" * 60)
        print(f"  Bot: @<seu_bot> (Chat ID: {TELEGRAM_CHAT_ID})")
        print(f"  Triggers criados para: temperatura, porta, água, energia")
        print(f"  Ação: 'Sala do servidor -> Telegram'")

    except Exception as e:
        print(f"\n❌ ERRO FATAL: {e}")
        sys.exit(1)
    finally:
        api.logout()


if __name__ == "__main__":
    main()

🎯 O Plano
O problema: o noc3d.html (dentro do iframe) não tem acesso aos dados do Zabbix, porque ele está isolado no iframe.

A solução: criar uma ponte via API do Zabbix dentro do noc3d.html. O iframe vai fazer requisições HTTP para a API do Zabbix (que já está rodando em localhost:8080), pegar os dados dos itens e atualizar o cena3d.js.

Por que isso funciona?

O iframe é servido pelo Grafana, mas executa no navegador.

O navegador pode fazer fetch() para http://localhost:8080/api_jsonrpc.php.

A API do Zabbix responde com os dados dos itens.

O cena3d.js recebe os dados e atualiza a cena.

        const container = document.getElementById('three-container');
        const badge = document.getElementById('badge');
        const cena = criarCena(container);

        // ============================================================
        // CONEXÃO COM A API DO ZABBIX

        const ZABBIX_URL = 'http://localhost:8080/api_jsonrpc.php';

        const ZABBIX_PASSWORD = 'zabbix';

        async function zabbixLogin() {
            const r = await fetch(ZABBIX_URL, {
                method: 'POST',
                headers: { 'Content-Type': 'application/json' },
                body: JSON.stringify({
                    jsonrpc: '2.0',
                    method: 'user.login',
                    params: { username: ZABBIX_USER, password: ZABBIX_PASSWORD },
                    id: 1,
                }),
            });
            const data = await r.json();
            if (data.error) throw new Error(data.error.message);
            authToken = data.result;

        async function zabbixGetHost() {
            const r = await fetch(ZABBIX_URL, {
                method: 'POST',
                headers: { 'Content-Type': 'application/json' },
                body: JSON.stringify({
                    jsonrpc: '2.0',
                    method: 'host.get',
                    params: { filter: { host: [HOST_NAME] } },
                    auth: authToken,
                    id: 2,
                }),
            });
            const data = await r.json();
            if (data.error) throw new Error(data.error.message);
            hostId = data.result[0].hostid;

            const keys = [
                'esp32.temp', 'esp32.umid', 'esp32.porta', 'esp32.modo',
                'esp32.estado', 'esp32.distancia', 'esp32.invasoes',
                'esp32.ambiente', 'esp32.luz', 'esp32.agua', 'esp32.energia'
            ];
            const r = await fetch(ZABBIX_URL, {
                method: 'POST',
                headers: { 'Content-Type': 'application/json' },
                body: JSON.stringify({
                    jsonrpc: '2.0',
                    method: 'item.get',
                    params: {
                        hostids: hostId,
                        output: ['key_', 'lastvalue'],
                        filter: { key_: keys },
                    },
                    auth: authToken,
                    id: 3,
                }),
            });
            const data = await r.json();
            if (data.error) throw new Error(data.error.message);
            return data.result;
        }

        function montarDados(items) {
            const map = {};
            for (const it of items) map[it.key_] = it.lastvalue;

            const temp = parseFloat(map['esp32.temp']);
            const umid = parseFloat(map['esp32.umid']);
            const porta = map['esp32.porta'] === 'true' || map['esp32.porta'] === '1' ? 1 : 0;
            const agua = map['esp32.agua'] === 'true' || map['esp32.agua'] === '1' ? 1 : 0;
            const energia = map['esp32.energia'] === 'true' || map['esp32.energia'] === '1' ? 1 : 0;

            return {
                ambiente: {
                    online: true,
                    energia: isNaN(energia) ? 1 : energia,
                    agua: isNaN(agua) ? 0 : agua,
                    porta: porta,
                    temperatura: isNaN(temp) ? null : temp,
                    umidade: isNaN(umid) ? null : umid,
                },
                servidores: [
                    { id: 'srv1', nome: 'SRV-01', estado: 'ok', cpu: 30 }
                ],
            };
        }

            try {
                if (!authToken) {
                    await zabbixLogin();
                    await zabbixGetHost();

                const items = await zabbixGetItems();
                const dados = montarDados(items);
                cena.atualizar(dados);
                const porta = dados.ambiente.porta === 1 ? 'porta aberta' : 'porta fechada';

                badge.textContent = `Erro: ${e.message}`;
                authToken = null; // força relogin na próxima tentativa
            }
            setTimeout(loop, 5000);
        }

        loop();
    </script>

🛠️ Passo 2 — Verificar se o CORS está liberado no Zabbix
O navegador faz fetch() para http://localhost:8080, mas o iframe é servido de http://localhost:3000. Isso é origem cruzada (CORS). O Zabbix precisa permitir isso.

Solução: adicionar o cabeçalho Access-Control-Allow-Origin: * na resposta da API do Zabbix. Isso é feito no nginx do container zabbix-web.

Alternativa mais simples: usar o proxy reverso do Grafana para acessar a API do Zabbix. Em vez de http://localhost:8080, usamos /api/zabbix (que o Grafana redireciona para o Zabbix). Mas isso exige configuração extra no Grafana.

Solução mais simples ainda: substituir o fetch direto por um proxy local no noc3d.html (usando XMLHttpRequest com withCredentials), mas isso também pode ser bloqueado.

A solução definitiva: usar mode: 'no-cors' no fetch (mas isso limita a leitura da resposta).

Vamos testar primeiro. Se o CORS bloquear, a gente resolve.

🎯 Próximo passo — Conectar o 3D aos dados reais
Agora que temos um "ponto seguro", podemos avançar com segurança. O próximo passo é substituir os dados mockados do noc3d.html pelos dados reais do Zabbix.

O problema: o iframe é servido pelo Grafana (localhost:3000), mas a API do Zabbix está em localhost:8080. Isso é origem cruzada (CORS), e o navegador pode bloquear o fetch().

A solução: usar o proxy reverso do Grafana para acessar a API do Zabbix. Ou usar mode: 'no-cors' no fetch (mas isso limita a leitura da resposta).

Vou te dar 3 opções:

Opção	Como funciona	Prós	Contras
A — Proxy reverso do Grafana	Adicionar um proxy no grafana.ini que redireciona /api/zabbix para http://zabbix-web:8080	CORS resolvido, sem mudar o Zabbix	Configuração extra no Grafana
B — CORS no Zabbix	Adicionar Access-Control-Allow-Origin: * no nginx do Zabbix	Simples, sem mudar o Grafana	Precisa editar o nginx do container
C — Mock + variáveis	Manter o mock, mas permitir testar com dados reais via parâmetros na URL	Simples, sem CORS	Não é tempo real
Minha recomendação: Opção A (proxy reverso do Grafana). É a mais limpa e não mexe no Zabbix.

📊 Avaliação Completa do Projeto monitoramento-cpd
Antes de qualquer código novo, uma auditoria honesta. Vou dividir em pontos fortes (o que faz o projeto valer a pena como referência) e campos de melhoria (o que precisa amadurecer para virar fonte de pesquisa para projetos maiores).

✅ Pontos Fortes
1. Arquitetura em camadas bem separada
O projeto tem 5 camadas claras com fronteiras definidas:

text
[ ESP32/Wokwi ] → [ MQTT ] → [ Mosquitto (bridge) ] → [ Zabbix ] → [ Grafana (2D + 3D) ]
Cada camada pode ser substituída sem quebrar as outras:

Trocar o Wokwi por hardware real → só muda o broker no firmware.

Trocar o Mosquitto por outro broker → só muda a config da bridge.

Trocar o Grafana por outro visualizador → a API do Zabbix continua lá.

Por que isso importa: é exatamente o que se espera de um projeto "fonte de pesquisa para projetos maiores". A arquitetura é o esqueleto reutilizável.

2. Filosofia IaC levada a sério
Você tem três níveis de IaC:

Nível	Arquivo	O que faz
Infra	docker-compose.yml	Sobe a stack inteira
Zabbix	configurar_zabbix.py	Cria host, itens, macros, template
Grafana	cpd-dashboard.json + cpd-3d.json + provisioning/*.yaml	Provisiona datasource e dashboards
Isso é raro em projetos de laboratório. A maioria configura tudo na interface e depois não consegue reproduzir. O seu é reproduzível por código.

Lição documentada: o script do Zabbix é idempotente (roda N vezes sem duplicar). Isso é o padrão profissional.

3. Documentação da jornada real (não da versão idealizada)
Os 12 arquivos em docs/ capturam:

Os erros reais (zsh, CORS, CSP, trends, hostid).

As decisões e por que foram tomadas.

As alternativas descartadas.

Isso é o que diferencia um tutorial de sucesso de um registro de engenharia. Quem ler o docs/ vai aprender o porquê, não só o como.

Sugestão de leitura para o README: o README deve apontar para docs/ como "bastidores da jornada". A documentação não é o README; é a história por trás dele.

4. NOC 3D com visual de operação (não de videogame)
O cena3d.js (original) tem decisões de UX muito boas:

Câmera ortográfica isométrica (estilo planta técnica, não FPS).

Cores de estado só quando significam algo.

Alarmes visuais (giroflex, calor, água, energia).

Cada problema reage diferente (tag efeito).

Isso mostra maturidade visual. Não é "3D por ser bonito". É 3D porque comunica estado operacional.

5. Tratamento de falhas no firmware (v8.3)
O firmware do ESP32 tem:

HMAC + nonce (anti-replay).

Watchdog.

Fila store-and-forward (MQTT offline).

Warm-up global.

Respostas de comando autenticadas.

Isso é segurança e confiabilidade acima da média para um projeto de laboratório. É um bom ponto de partida para estudar IoT seguro.

🔧 Campos de Melhoria
1. ⚠️ Falta de "casca" de projeto reutilizável
Hoje, se alguém quiser reproduzir o projeto, precisa:

Entender o docker-compose.yml inteiro.

Ler 12 arquivos de documentação.

Copiar e colar comandos soltos.

Falta um Makefile ou install.sh que faça o setup em uma linha:

bash
make setup    # cria .env, sobe stack, roda configurar_zabbix.py
make up       # sobe tudo
make down     # para tudo
make reset    # apaga volumes e recomeça
Por que importa: para virar "fonte de pesquisa", o projeto precisa ter uma porta de entrada óbvia. Hoje, a porta é "leia o README e reze".

2. ⚠️ Falta de testes automatizados
O configurar_zabbix.py é idempotente, mas não tem teste unitário. Se alguém quebrar a lógica de upsert, só descobre quando roda e vê [!].

Sugestão mínima:

tests/test_configurar_zabbix.py com pytest e mocks da API do Zabbix.

Verifica que rodar 2x não duplica.

Verifica que cada tipo de item (float, text, unsigned) usa trends correto.

Por que importa: sem teste, o código não é confiável para "projetos maiores". É só "funciona na minha máquina".

3. ⚠️ Documentação espalhada, sem índice navegável
Os 12 arquivos .txt em docs/ são um tesouro, mas estão soltos. O leitor precisa abrir um por um para entender a jornada.

Sugestão:

Renomear para docs/01-arquitetura.md, docs/02-passos-0.md, etc.

Adicionar um docs/00-indice.md com links e resumos.

O README.md aponta para o índice.

Por que importa: para virar referência de pesquisa, a documentação precisa ser navegável, não só completa.

4. ⚠️ Falta de "contrato" do firmware
O firmware v8.3 tem MUITA funcionalidade (HMAC, OTA, RFID, watchdog, sensores). Mas não há um contrato explícito entre ele e o Zabbix:

Quais campos ele publica no MQTT?

Quais comandos ele aceita?

Quais garantias ele oferece (latência, formato, autenticação)?

Sugestão:

Criar docs/protocolo-mqtt.md documentando todos os tópicos e payloads.

Criar um JSON Schema (firmware/status-schema.json) que valide o JSON de status.

O configurar_zabbix.py lê esse schema para criar os itens dependentes automaticamente.

Por que importa: para virar referência, o protocolo precisa ser explícito e versionado. Se o firmware mudar, o contrato muda junto.

5. ⚠️ NOC 3D não conecta aos dados reais (ainda)
Hoje, o noc3d.html usa dados mockados. Isso é bom para desenvolvimento, mas o NOC 3D não é "de verdade" até buscar do Zabbix.

Sugestão:

Implementar a Opção A (proxy reverso do Grafana) para evitar CORS.

Ou usar Server-Sent Events (SSE) no Grafana (que já tem suporte a Live) e expor os dados do Zabbix em tempo real.

Documentar a decisão no docs/.

Por que importa: o NOC 3D precisa reagir de verdade para valer a pena. Mock é para desenvolvimento; real é para operação.

6. ⚠️ Falta de observabilidade do próprio pipeline
O pipeline inteiro (Wokwi → Mosquitto → Zabbix → Grafana) não tem métricas de si mesmo:

Quantas mensagens MQTT chegam por segundo?

Qual é a latência entre o publish e o latest data?

Quantas vezes o Zabbix falhou em ler o Mosquitto?

Sugestão:

Adicionar Prometheus + Grafana para monitorar o próprio pipeline.

Ou usar healthchecks no docker-compose.yml que expõem /metrics para cada serviço.

Por que importa: para virar referência, o projeto precisa mostrar como observar o observador. É o próximo nível.

7. ⚠️ Falta de "estratégia de backup e recuperação"
O projeto tem volumes do Docker (Postgres, Grafana, Mosquitto), mas não tem:

Script de backup (backup.sh).

Script de restauração (restore.sh).

Documentação de "o que fazer se o Postgres quebrar".

Sugestão:

scripts/backup.sh que faz pg_dump e salva em backups/.

scripts/restore.sh que restaura de um .sql.

docs/backup.md explicando a estratégia.

Por que importa: para projetos maiores, backup e recuperação são não-negociáveis. Um laboratório sem backup é um laboratório que ensina errado.

8. ⚠️ Falta de "documentação de decisões arquiteturais" (ADRs)
Você tem decisões importantes (broker local vs. público, bridge vs. HTTP, 3D vs. 2D) espalhadas nos .txt. Mas não há um ADR (Architecture Decision Record) formal.

Sugestão:

docs/adr/001-broker-local.md

docs/adr/002-bridge-mqtt.md

docs/adr/003-3d-via-iframe.md

etc.

Cada ADR tem: Contexto, Decisão, Consequências, Alternativas descartadas.

Por que importa: para projetos maiores, a história das decisões é mais importante que o código. Um ADR explica por que o código é assim.

9. ⚠️ Falta de "roadmap explícito" pós-fase-6
Você tem as fatias 0–6 e a 7 (Ações HMAC), mas não há:

Backlog do que vem depois (ex: multi-rack, multi-CPD).

Diagrama de evolução (o que vira o quê quando o projeto crescer).

Como adicionar um segundo ESP32 ou um segundo Zabbix.

Sugestão:

docs/roadmap.md com as próximas fases (curto, médio, longo prazo).

docs/como-adicionar-sala.md (passo a passo para replicar o projeto em outro CPD).

Por que importa: para virar fonte de pesquisa, o projeto precisa mostrar como escalar sem reescrever tudo.

10. ⚠️ Falta de licença explícita
O LICENSE-three.txt está lá (para o Three.js), mas o projeto em si não tem licença.

Sugestão:

Adicionar LICENSE (MIT, Apache 2.0, GPL — depende do objetivo).

Adicionar um CONTRIBUTING.md com regras para contribuições.

Adicionar um CODE_OF_CONDUCT.md (opcional, mas recomendado).

Por que importa: sem licença, o projeto não pode ser usado como base para projetos maiores (juridicamente).
